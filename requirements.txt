# Core runtime pins for a plain `pip install -r requirements.txt` dev
# setup. Every pin is exact: the image installs the hashed lockfiles in
# requirements/ instead, and these versions must equal the lockfiles'
# public versions (no +cpu/+cu132 local segment), enforced by
# `test_requirements_match_the_lockfiles`. Bump a version here only by
# recompiling the locks (see requirements/server.cpu.in) and copying
# the result back.
#
# PyTorch is installed by the Dockerfile from the TORCH_INDEX_URL build
# arg so the build (CPU vs CUDA) can be chosen; these pins keep dev
# installs working.
#
# The FastAPI server stack (fastapi/uvicorn/websockets/pydantic) is intentionally
# absent: it is now the base dependency set of the `spikeforge-server`
# distribution, so core stays headless.
torch==2.14.1
torchvision==0.29.1
snntorch==1.0.0
matplotlib==3.11.2
Pillow==13.0.0
numpy==2.5.4
psutil==7.2.2
